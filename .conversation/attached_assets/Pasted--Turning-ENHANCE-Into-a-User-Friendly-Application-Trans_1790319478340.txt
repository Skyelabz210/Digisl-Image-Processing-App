# Turning ENHANCE! Into a User-Friendly Application

### Step 2: Create the Application Structure

Here's a complete Tkinter application that incorporates your ENHANCE! code:

```python
import tkinter as tk
from tkinter import filedialog, Scale, Label, Button, Frame
from PIL import Image, ImageTk
import cv2
import numpy as np
import matplotlib.pyplot as plt
from matplotlib.backends.backend_tkagg import FigureCanvasTkAgg
import os

class EnhanceApp:
    def __init__(self, root):
        self.root = root
        self.root.title("ENHANCE! Image Processor")
        self.root.geometry("1200x800")
        
        # Variables
        self.image_path = None
        self.original_image = None
        self.entropy_threshold = 5.0
        self.enhancement_strength = 1.5
        self.results = None
        
        # Create UI
        self.create_ui()
    
    def create_ui(self):
        # Top frame for controls
        control_frame = Frame(self.root)
        control_frame.pack(side=tk.TOP, fill=tk.X, padx=10, pady=10)
        
        # Load image button
        load_btn = Button(control_frame, text="Load Image", command=self.load_image)
        load_btn.pack(side=tk.LEFT, padx=10)
        
        # Entropy threshold slider
        threshold_frame = Frame(control_frame)
        threshold_frame.pack(side=tk.LEFT, padx=20)
        
        Label(threshold_frame, text="Entropy Threshold:").pack(anchor=tk.W)
        self.threshold_slider = Scale(threshold_frame, from_=1.0, to=10.0, resolution=0.1,
                                     orient=tk.HORIZONTAL, length=200, 
                                     command=self.update_threshold)
        self.threshold_slider.set(self.entropy_threshold)
        self.threshold_slider.pack()
        
        # Enhancement strength slider
        strength_frame = Frame(control_frame)
        strength_frame.pack(side=tk.LEFT, padx=20)
        
        Label(strength_frame, text="Enhancement Strength:").pack(anchor=tk.W)
        self.strength_slider = Scale(strength_frame, from_=1.0, to=3.0, resolution=0.1,
                                    orient=tk.HORIZONTAL, length=200,
                                    command=self.update_strength)
        self.strength_slider.set(self.enhancement_strength)
        self.strength_slider.pack()
        
        # Process button
        process_btn = Button(control_frame, text="Process Image", command=self.process_image)
        process_btn.pack(side=tk.LEFT, padx=20)
        
        # Save button
        save_btn = Button(control_frame, text="Save Enhanced Image", command=self.save_image)
        save_btn.pack(side=tk.LEFT, padx=10)
        
        # Main display area
        self.display_frame = Frame(self.root)
        self.display_frame.pack(fill=tk.BOTH, expand=True, padx=10, pady=10)
        
        # Status label
        self.status_label = Label(self.root, text="Load an image to begin", bd=1, relief=tk.SUNKEN, anchor=tk.W)
        self.status_label.pack(side=tk.BOTTOM, fill=tk.X)
    
    def load_image(self):
        self.image_path = filedialog.askopenfilename(
            filetypes=[("Image files", "*.jpg *.jpeg *.png *.bmp *.tif")])
        
        if not self.image_path:
            return
        
        self.status_label.config(text=f"Loaded: {os.path.basename(self.image_path)}")
        self.original_image = cv2.imread(self.image_path)
        self.original_image = cv2.cvtColor(self.original_image, cv2.COLOR_BGR2RGB)
        
        # Display original image
        self.display_original_image()
    
    def display_original_image(self):
        # Clear display frame
        for widget in self.display_frame.winfo_children():
            widget.destroy()
        
        # Create a figure for the original image
        fig = plt.Figure(figsize=(6, 4), dpi=100)
        ax = fig.add_subplot(111)
        ax.imshow(self.original_image)
        ax.set_title("Original Image")
        ax.axis('off')
        
        # Embed the figure in the tkinter window
        canvas = FigureCanvasTkAgg(fig, master=self.display_frame)
        canvas.draw()
        canvas.get_tk_widget().pack(side=tk.TOP, fill=tk.BOTH, expand=True)
    
    def update_threshold(self, value):
        self.entropy_threshold = float(value)
    
    def update_strength(self, value):
        self.enhancement_strength = float(value)
    
    def process_image(self):
        if self.original_image is None:
            self.status_label.config(text="Please load an image first")
            return
        
        self.status_label.config(text="Processing image...")
        self.root.update()
        
        # Process the image
        self.results = self.enhance_image(self.original_image, 
                                         self.entropy_threshold, 
                                         self.enhancement_strength)
        
        # Display results
        self.display_results()
        self.status_label.config(text="Processing complete")
    
    def display_results(self):
        # Clear display frame
        for widget in self.display_frame.winfo_children():
            widget.destroy()
        
        # Create a figure with subplots
        fig = plt.Figure(figsize=(12, 10), dpi=100)
        
        # Original image
        ax1 = fig.add_subplot(221)
        ax1.imshow(self.results['original'])
        ax1.set_title('Original Image')
        ax1.axis('off')
        
        # Entropy map
        ax2 = fig.add_subplot(222)
        entropy_img = ax2.imshow(self.results['entropy_map'], cmap='hot')
        ax2.set_title('Entropy Map')
        ax2.axis('off')
        fig.colorbar(entropy_img, ax=ax2, fraction=0.046, pad=0.04)
        
        # Enhancement mask
        ax3 = fig.add_subplot(223)
        mask_img = ax3.imshow(self.results['enhancement_mask'], cmap='gray')
        ax3.set_title('Enhancement Mask')
        ax3.axis('off')
        fig.colorbar(mask_img, ax=ax3, fraction=0.046, pad=0.04)
        
        # Enhanced image
        ax4 = fig.add_subplot(224)
        ax4.imshow(self.results['enhanced'])
        ax4.set_title('Enhanced Image')
        ax4.axis('off')
        
        # Adjust layout
        fig.tight_layout()
        
        # Embed the figure in the tkinter window
        canvas = FigureCanvasTkAgg(fig, master=self.display_frame)
        canvas.draw()
        canvas.get_tk_widget().pack(side=tk.TOP, fill=tk.BOTH, expand=True)
    
    def save_image(self):
        if self.results is None:
            self.status_label.config(text="Process an image before saving")
            return
        
        save_path = filedialog.asksaveasfilename(
            defaultextension=".png",
            filetypes=[("PNG files", "*.png"), ("JPEG files", "*.jpg"), ("All files", "*.*")])
        
        if not save_path:
            return
        
        # Save the enhanced image
        enhanced_image = self.results['enhanced']
        enhanced_image_bgr = cv2.cvtColor(enhanced_image, cv2.COLOR_RGB2BGR)
        cv2.imwrite(save_path, enhanced_image_bgr)
        
        self.status_label.config(text=f"Enhanced image saved as {os.path.basename(save_path)}")
    
    def enhance_image(self, image, entropy_threshold=5.0, enhancement_strength=1.5):
        """
        A simplified ENHANCE! implementation that uses entropy to guide image enhancement.
        
        Args:
            image: Input image array
            entropy_threshold: Threshold for identifying high-entropy regions
            enhancement_strength: How strongly to enhance high-entropy regions
            
        Returns:
            Dictionary containing the original image, entropy map, and enhanced image
        """
        # Create grayscale version for entropy calculation
        gray = cv2.cvtColor(image, cv2.COLOR_RGB2GRAY)
        
        # Calculate entropy map
        entropy_map = np.zeros_like(gray, dtype=float)
        window_size = 9  # Must be odd
        half_window = window_size // 2
        
        # Pad the image to handle borders
        padded = cv2.copyMakeBorder(gray, half_window, half_window, half_window, half_window, 
                                   cv2.BORDER_REFLECT)
        
        # Calculate entropy for each pixel's neighborhood
        for i in range(half_window, padded.shape[0] - half_window):
            for j in range(half_window, padded.shape[1] - half_window):
                # Extract window
                window = padded[i-half_window:i+half_window+1, j-half_window:j+half_window+1]
                
                # Calculate histogram
                hist, _ = np.histogram(window, bins=256, range=[0, 256], density=True)
                hist = hist[hist > 0]  # Remove zero probabilities
                
                # Calculate entropy
                entropy = -np.sum(hist * np.log2(hist))
                
                # Store entropy value
                entropy_map[i-half_window, j-half_window] = entropy
        
        # Normalize entropy map for visualization
        entropy_norm = cv2.normalize(entropy_map, None, 0, 1, cv2.NORM_MINMAX)
        
        # Create enhancement mask
        enhancement_mask = (entropy_map > entropy_threshold).astype(float)
        
        # Smooth the mask to avoid hard transitions
        enhancement_mask = cv2.GaussianBlur(enhancement_mask, (15, 15), 0)
        
        # Convert to 3-channel for multiplication
        enhancement_mask_3ch = np.stack([enhancement_mask] * 3, axis=2)
        
        # Enhance the image using a simple sharpening operator
        kernel = np.array([[-1, -1, -1],
                           [-1,  9, -1],
                           [-1, -1, -1]])
        sharpened = cv2.filter2D(image, -1, kernel)
        
        # Apply sharpening only in high-entropy regions
        strength_factor = enhancement_strength - 1.0  # Convert to additive factor
        enhancement = image + (sharpened - image) * enhancement_mask_3ch * strength_factor
        enhancement = np.clip(enhancement, 0, 255).astype(np.uint8)
        
        return {
            'original': image,
            'entropy_map': entropy_norm,
            'enhanced': enhancement,
            'enhancement_mask': enhancement_mask
        }

if __name__ == "__main__":
    root = tk.Tk()
    app = EnhanceApp(root)
    root.mainloop()
```

def save_results(results, session_id):
    """Save the result images to disk"""
    # Create a directory for this session
    session_dir = os.path.join(app.config['RESULTS_FOLDER'], session_id)
    os.makedirs(session_dir, exist_ok=True)
    
    # Save enhanced image
    enhanced_path = os.path.join(session_dir, 'enhanced.png')
    cv2.imwrite(enhanced_path, cv2.cvtColor(results['enhanced'], cv2.COLOR_RGB2BGR))
    
    # Create and save visualization
    fig, axes = plt.subplots(2, 2, figsize=(12, 10))
    
    axes[0, 0].imshow(results['original'])
    axes[0, 0].set_title('Original Image')
    axes[0, 0].axis('off')
    
    axes[0, 1].imshow(results['entropy_map'], cmap='hot')
    axes[0, 1].set_title('Entropy Map')
    axes[0, 1].axis('off')
    
    axes[1, 0].imshow(results['enhancement_mask'], cmap='gray')
    axes[1, 0].set_title('Enhancement Mask')
    axes[1, 0].axis('off')
    
    axes[1, 1].imshow(results['enhanced'])
    axes[1, 1].set_title('Enhanced Image')
    axes[1, 1].axis('off')
    
    plt.tight_layout()
    
    # Save visualization
    viz_path = os.path.join(session_dir, 'visualization.png')
    plt.savefig(viz_path, dpi=100)
    plt.close()
    
    return {
        'enhanced': os.path.join('results', session_id, 'enhanced.png'),
        'visualization': os.path.join('results', session_id, 'visualization.png')
    }

@app.route('/', methods=['GET', 'POST'])
def index():
    if request.method == 'POST':
        # Check if a file was uploaded
        if 'file' not in request.files:
            return redirect(request.url)
        
        file = request.files['file']
        
        # If user doesn't select a file
        if file.filename == '':
            return redirect(request.url)
        
        # If file is valid
        if file:
            # Generate a unique session ID
            session_id = str(uuid.uuid4())
            
            # Save the uploaded file
            filename = f"{session_id}_{file.filename}"
            file_path = os.path.join(app.config['UPLOAD_FOLDER'], filename)
            file.save(file_path)
            
            # Get parameters
            entropy_threshold = float(request.form.get('entropy_threshold', 5.0))
            enhancement_strength = float(request.form.get('enhancement_strength', 1.5))
            
            # Process the image
            results = enhance_image(file_path, entropy_threshold, enhancement_strength)
            
            # Save results
            paths = save_results(results, session_id)
            
            # Return results page
            return render_template('results.html', 
                                  original=os.path.join('uploads', filename),
                                  enhanced=paths['enhanced'],
                                  visualization=paths['visualization'],
                                  session_id=session_id)
    
    # GET request - show upload form
    return render_template('index.html')

if __name__ == '__main__':
    app.run(debug=True)

        'entropy_map': entropy_norm,

st.title("ENHANCE! Image Processor")
st.write("Upload an image to enhance details using entropy-guided processing.")

# Sidebar for controls
st.sidebar.header("Settings")
entropy_threshold = st.sidebar.slider("Entropy Threshold", 1.0, 10.0, 5.0, 0.1)

# File uploader

if uploaded_file is not None:
    # Convert uploaded file to OpenCV format
    file_bytes = np.asarray(bytearray(uploaded_file.read()), dtype=np.uint8)
    image = cv2.imdecode(file_bytes, cv2.IMREAD_COLOR)
    image = cv2.cvtColor(image, cv2.COLOR_BGR2RGB)
    
    # Process button
    if st.button("Process Image"):
        with st.spinner("Processing image..."):
            # Process the image
            results = enhance_image(image, entropy_threshold, enhancement_strength)
            
            # Display results
            col1, col2 = st.columns(2)
            
            with col1:
                st.subheader("Original Image")
                st.image(results['original'], channels="RGB", use_column_width=True)
                
                st.subheader("Enhancement Mask")
                st.image(results['enhancement_mask'], use_column_width=True)
            
            with col2:
                st.subheader("Entropy Map")
                st.image(results['entropy_map'], use_column_width=True, clamp=True)
                
                st.subheader("Enhanced Image")
                st.image(results['enhanced'], channels="RGB", use_column_width=True)
            
            # Download button for enhanced image
            enhanced_pil = Image.fromarray(results['enhanced'])
            with tempfile.NamedTemporaryFile(delete=False, suffix='.png') as tmp_file:
                enhanced_pil.save(tmp_file.name)
                st.download_button(
                    label="Download Enhanced Image",
                    data=open(tmp_file.name, 'rb').read(),
                    file_name="enhanced_image.png",
                    mime="image/png"
                )
                # Clean up temp file
                os.unlink(tmp_file.name)
    else:
        # Show original image before processing
        st.subheader("Original Image")
        st.image(image, channels="RGB", use_column_width=True)

## Comparison of Options

1. **Tkinter Desktop App**
   - Pros: Works offline, standalone executable, simple distribution
   - Cons: More complex UI code, less polished appearance

2. **Flask Web App**
   - Pros: Accessible from any device with a browser, clean interface
   - Cons: Requires server deployment, more complex setup

3. **Streamlit App**
   - Pros: Extremely easy to create, clean modern interface, quick deployment
   - Cons: Less customization, requires Python environment to run locally

## Improving Your App

Whichever option you choose, here are some ways to enhance your application:

1. **Add More Enhancement Options**
   - Different sharpening algorithms
   - Color enhancement features
   - Noise reduction options

2. **Optimize Performance**
   - Use NumPy vectorization for entropy calculation
   - Add a progress indicator for large images
   - Add image resizing for very large uploads

3. **Improve the User Experience**
   - Add before/after comparison sliders
   - Include presets for different image types
   - Add tooltips explaining each parameter

## Distributing Your App

- **Desktop App**: Create an installer using a tool like Inno Setup (Windows) or create a DMG file (Mac)
- **Web App**: Deploy to a cloud service like Heroku, PythonAnywhere, or Google Cloud Run
- **Streamlit App**: Share via Streamlit Cloud, which offers free hosting for public apps

## Final Thought

The Streamlit option is likely the best balance of simplicity and functionality for most users. It requires minimal coding, produces a professional-looking interface, and can be easily shared online.
